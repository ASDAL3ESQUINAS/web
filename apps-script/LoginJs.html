document.getElementById('loginForm').addEventListener('submit',function(event){event.preventDefault();var button=this.querySelector('button');var message=document.getElementById('message');var pinField=document.getElementById('pinField');var documentNumber=document.getElementById('document').value;var pin=document.getElementById('pin').value;button.disabled=true;message.textContent='';var done=function(result){button.disabled=false;if(result.requiresPin){pinField.hidden=false;document.getElementById('pin').required=true;message.textContent=result.message;return;}if(!result.ok){message.textContent=result.message;return;}sessionStorage.setItem('asdalSession',result.token);window.top.location.href='?view=app';};var request=google.script.run.withSuccessHandler(done).withFailureHandler(function(){button.disabled=false;message.textContent='No fue posible validar el acceso.';});if(pinField.hidden)request.startLogin(documentNumber);else request.loginAdministrator(documentNumber,pin);});
